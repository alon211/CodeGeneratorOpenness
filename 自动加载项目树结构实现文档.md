### 3. 新增自动加载方法

#### 3.1 LoadPlcSoftwareStructures 方法

这是核心的自动加载方法，负责调用各个具体的加载方法：

```csharp
private void LoadPlcSoftwareStructures(PlcSoftware software, TreeNode root)
{
    Logger.LogInfo("开始自动加载PlcSoftware的所有结构组件", "LoadPlcSoftwareStructures");

    // 1. 加载程序块组 (BlockGroup)
    LoadBlockGroup(software, root);

    // 2. 加载数据类型组 (TypeGroup)
    LoadTypeGroup(software, root);

    // 3. 加载标签表组 (TagTableGroup)
    LoadTagTableGroup(software, root);

    // 4. 加载外部源文件组 (ExternalSourceGroup)
    LoadExternalSourceGroup(software, root);

    // 5. 加载PLC报警文本列表组 (PlcAlarmTextlistGroup)
    LoadPlcAlarmTextlistGroup(software, root);

    // 6. 加载技术对象组 (TechnologicalObjectGroup)
    LoadTechnologicalObjectGroup(software, root);

    // 7. 加载监视和强制表组 (WatchAndForceTableGroup)
    LoadWatchAndForceTableGroup(software, root);

    Logger.LogInfo("PlcSoftware结构组件加载完成", "LoadPlcSoftwareStructures");
}
```

为每个 `PlcSoftware` 结构组件创建了专门的加载方法：

```csharp
private void LoadSoftwareUnits(PlcSoftware software, TreeNode root)
{
    try
    {
        Logger.LogInfo("开始加载软件单元组", "LoadSoftwareUnits");
        
        // 通过GetService获取PlcUnitProvider
        PlcUnitProvider unitProvider = software.GetService<PlcUnitProvider>();
        
        if (unitProvider?.UnitGroup != null)
        {
            PlcUnitSystemGroup unitSystemGroup = unitProvider.UnitGroup;
            int unitCount = GetUnitCount(unitSystemGroup);
            
            if (unitCount > 0)
            {
                TreeNode unitsNode = new TreeNode($"Software Units ({unitCount})");
                unitsNode.ImageIndex = 8; // 使用文件夹图标
                unitsNode.SelectedImageIndex = 8;
                root.Nodes.Add(unitsNode);
                
                // 添加软件单元
                AddSoftwareUnits(unitSystemGroup, unitsNode);
                
                Logger.LogInfo($"软件单元组加载完成，共{unitCount}个单元", "LoadSoftwareUnits");
            }
            else
            {
                Logger.LogWarning("软件单元组为空", "LoadSoftwareUnits");
            }
        }
        else
        {
            Logger.LogWarning("无法获取软件单元提供者或单元组", "LoadSoftwareUnits");
        }
    }
    catch (Exception ex)
    {
        Logger.LogWarning($"软件单元组加载失败: {ex.Message}", "LoadSoftwareUnits");
    }
}
```

##### 3.4 递归解析软件单元组件方法

```csharp
private void AddSoftwareUnitComponents(object unit, TreeNode unitNode)
{
    try
    {
        Logger.LogInfo($"开始解析软件单元组件: {GetUnitName(unit)}", "AddSoftwareUnitComponents");
        
        // 使用反射获取PlcUnitBase的属性
        Type unitType = unit.GetType();
        
        // 检查BlockGroup属性
        var blockGroupProperty = unitType.GetProperty("BlockGroup");
        if (blockGroupProperty != null)
        {
            var blockGroup = blockGroupProperty.GetValue(unit);
            if (blockGroup != null)
            {
                int blockCount = GetBlockCount((dynamic)blockGroup);
                if (blockCount > 0)
                {
                    TreeNode blocksNode = new TreeNode($"Program Blocks ({blockCount})");
                    blocksNode.ImageIndex = 8;
                    blocksNode.SelectedImageIndex = 8;
                    unitNode.Nodes.Add(blocksNode);
                    AddPlcBlocks((dynamic)blockGroup, blocksNode);
                }
            }
        }
        
        // 检查TypeGroup属性
        var typeGroupProperty = unitType.GetProperty("TypeGroup");
        if (typeGroupProperty != null)
        {
            var typeGroup = typeGroupProperty.GetValue(unit);
            if (typeGroup != null)
            {
                int typeCount = GetTypeCount((dynamic)typeGroup);
                if (typeCount > 0)
                {
                    TreeNode typesNode = new TreeNode($"PLC Data types ({typeCount})");
                    typesNode.ImageIndex = 8;
                    typesNode.SelectedImageIndex = 8;
                    unitNode.Nodes.Add(typesNode);
                    AddPlcTypes((dynamic)typeGroup, typesNode);
                }
            }
        }
        
        // 检查TagTableGroup属性
        var tagTableGroupProperty = unitType.GetProperty("TagTableGroup");
        if (tagTableGroupProperty != null)
        {
            var tagTableGroup = tagTableGroupProperty.GetValue(unit);
            if (tagTableGroup != null)
            {
                AddTagTableGroup((dynamic)tagTableGroup, unitNode);
            }
        }
        
        // 检查ExternalSourceGroup属性
        var externalSourceGroupProperty = unitType.GetProperty("ExternalSourceGroup");
        if (externalSourceGroupProperty != null)
        {
            var externalSourceGroup = externalSourceGroupProperty.GetValue(unit);
            if (externalSourceGroup != null)
            {
                AddExternalSourceGroup((dynamic)externalSourceGroup, unitNode);
            }
        }
        
        // 检查PlcAlarmTextlistGroup属性
        var alarmTextlistGroupProperty = unitType.GetProperty("PlcAlarmTextlistGroup");
        if (alarmTextlistGroupProperty != null)
        {
            var alarmTextlistGroup = alarmTextlistGroupProperty.GetValue(unit);
            if (alarmTextlistGroup != null)
            {
                AddAlarmTextlists((dynamic)alarmTextlistGroup, unitNode);
            }
        }
        
        Logger.LogInfo($"软件单元组件解析完成: {GetUnitName(unit)}", "AddSoftwareUnitComponents");
    }
    catch (Exception ex)
    {
        Logger.LogWarning($"软件单元组件解析失败: {ex.Message}", "AddSoftwareUnitComponents");
    }
}
```

```csharp
private string GetUnitName(object unit)
{
    try
    {
        // 使用反射安全获取Name属性
        Type unitType = unit.GetType();
        var nameProperty = unitType.GetProperty("Name");
        if (nameProperty != null)
        {
            return nameProperty.GetValue(unit)?.ToString() ?? "Unknown";
        }
        return "Unknown";
    }
    catch
    {
        return "Unknown";
    }
}
```

#### 2. 图标加载实现

在 `frmMainForm.cs` 文件中新增了 `LoadAdditionalIcons()` 方法，用于从 `Ressources` 文件夹动态加载图标：

```csharp
private void LoadAdditionalIcons()
{
    try
    {
        string resourcesPath = Path.Combine(Application.StartupPath, "Ressources");
        
        // 加载变量表图标
        string tagTableIconPath = Path.Combine(resourcesPath, "tag_table.png");
        if (File.Exists(tagTableIconPath))
        {
            using (Image tagTableIcon = Image.FromFile(tagTableIconPath))
            {
                imageList1.Images.Add(tagTableIcon);
            }
            Logger.LogInfo("变量表图标加载成功", "LoadAdditionalIcons");
        }
        else
        {
            Logger.LogWarning($"变量表图标文件不存在: {tagTableIconPath}", "LoadAdditionalIcons");
        }
        
        // 加载软件单元图标
        string softwareUnitsIconPath = Path.Combine(resourcesPath, "softwareUnits.png");
        if (File.Exists(softwareUnitsIconPath))
        {
            using (Image softwareUnitsIcon = Image.FromFile(softwareUnitsIconPath))
            {
                imageList1.Images.Add(softwareUnitsIcon);
            }
            Logger.LogInfo("软件单元图标加载成功", "LoadAdditionalIcons");
        }
        else
        {
            Logger.LogWarning($"软件单元图标文件不存在: {softwareUnitsIconPath}", "LoadAdditionalIcons");
        }
    }
    catch (Exception ex)
    {
        Logger.LogError($"加载附加图标时发生错误: {ex.Message}", "LoadAdditionalIcons");
    }
}
```
